// ==UserScript==
// @name         CSGO Auto Clicker
// @namespace    eek
// @version      0.2
// @description  Update: Automatically clicks the "Accept" button in caseclicker by pressing button "s" on your keyboard, to turn off the auto clicker press "s" again.
// @author       King Jayden
// @match        https://kingofkfcjamal.github.io/*
// @grant        none
// @downloadURL https://update.greasyfork.org/scripts/372597/CSGO%20Clicker%20Money%20Hack%20-%20CSGO%20CSGOClicker%20Case%20CaseClicker%20Clicker.user.js
// @updateURL https://update.greasyfork.org/scripts/372597/CSGO%20Clicker%20Money%20Hack%20-%20CSGO%20CSGOClicker%20Case%20CaseClicker%20Clicker.meta.js
// ==/UserScript==

(function() {
    'use strict';

    let myInterval = null;

    // Listen for key presses on the page
    window.addEventListener('keydown', function(event) {
        // Change 's' to any lowercase letter you prefer
        if (event.key === 's') {
            if (!myInterval) {
                // Start clicking if it is stopped
                myInterval = setInterval(function() {
                    const button = document.getElementById("acceptButton");
                    if (button) button.click();
                }, 1);
                console.log("Autoclicker STARTED");
            } else {
                // Stop clicking if it is running
                clearInterval(myInterval);
                myInterval = null;
                console.log("Autoclicker STOPPED");
            }
        }
    });
})();
